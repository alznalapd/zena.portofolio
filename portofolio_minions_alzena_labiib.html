sketch_minion_monitoring.ino (C/C++) 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Servo.h>

// Bello! Inisialisasi Minion Hardware Pin
const int LED_PIN = 13;
const int DHT_PIN = A0;
LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo myServo;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  lcd.init();
  lcd.backlight();
  myServo.attach(9);
  
  lcd.setCursor(0, 0);
  lcd.print("BELLO! Minion IoT");
}

void loop() {
  // Membaca sensor suhu & update tampilan LCD
  int sensorVal = analogRead(DHT_PIN);
  float temp = sensorVal * (5.0 / 1023.0) * 100.0;
  
  if (temp > 35.0) {
    digitalWrite(LED_PIN, HIGH); // Peringatan overheat
  }
  delay(500);
}
